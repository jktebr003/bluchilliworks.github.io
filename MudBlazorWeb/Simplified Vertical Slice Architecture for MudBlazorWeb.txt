Simplified Vertical Slice Architecture for MudBlazorWeb
--------------------------------------------------------

Project: MudBlazorWeb (Blazor WebAssembly + .NET 9)
Architecture: Vertical Slice + Clean Architecture (Simplified)
---

Overview
This is a simplified folder structure that maintains Clean Architecture principles while reducing folder depth and complexity. Each feature is self-contained with clear boundaries.


Key Simplifications
1.	Merged Api + Application: DTOs and handlers live together in Application/
2.	Fluxor in UI root: State management directly under UI/ (no nested State/ folder)
3.	Flatten Components: Components directly in UI/ (no nested Components/ folder)
4.	Single DI file: One DependencyInjection.cs per feature (at feature root)
5.	Tests at feature root: Tests/ directly under feature folder
6.	Command + Handler in same file: Related code stays together
---

Layer Responsibilities
1. Application Layer
Location: Application/
Contains: Commands, Queries, Handlers, DTOs, Validation
Pattern: Combine Command/Query with Handler in one file

using MediatR;

namespace MudBlazorWeb.Features.Pricing.Application;

// DTO
public record PricingDto(string PlanType, decimal Price, List<string> Features);

// Query
public record GetPricingsQuery : IRequest<Result<List<PricingDto>>>;

// Handler (same file)
public class GetPricingsHandler : IRequestHandler<GetPricingsQuery, Result<List<PricingDto>>>
{
    private readonly IPricingRepository _repository;

    public GetPricingsHandler(IPricingRepository repository)
    {
        _repository = repository;
    }

    public async Task<Result<List<PricingDto>>> Handle(
        GetPricingsQuery request, 
        CancellationToken cancellationToken)
    {
        var pricings = await _repository.GetAllAsync(cancellationToken);
        
        var dtos = pricings
            .Select(p => new PricingDto(p.PlanType, p.Price, p.Features))
            .ToList();
        
        return Result<List<PricingDto>>.Success(dtos);
    }
}

// Endpoints (same file for minimal API)
using MediatR;
using Microsoft.AspNetCore.Builder;
using Microsoft.AspNetCore.Http;
using Microsoft.AspNetCore.Routing;

namespace MudBlazorWeb.Features.Pricing.Application;

public static class PricingEndpoints
{
    public static IEndpointRouteBuilder MapPricingEndpoints(this IEndpointRouteBuilder endpoints)
    {
        var group = endpoints.MapGroup("/api/pricing")
            .WithTags("Pricing");

        group.MapGet("/", GetPricings)
            .WithName("GetPricings")
            .Produces<List<PricingDto>>();

        group.MapGet("/{id}", GetPricingById)
            .WithName("GetPricingById")
            .Produces<PricingDto>()
            .Produces(StatusCodes.Status404NotFound);

        group.MapPost("/", CreatePricing)
            .WithName("CreatePricing")
            .Produces<Guid>(StatusCodes.Status201Created)
            .ProducesValidationProblem();

        return endpoints;
    }

    private static async Task<IResult> GetPricings(IMediator mediator)
    {
        var result = await mediator.Send(new GetPricingsQuery());
        
        return result.IsSuccess 
            ? Results.Ok(result.Value) 
            : Results.BadRequest(result.ErrorMessage);
    }

    private static async Task<IResult> GetPricingById(Guid id, IMediator mediator)
    {
        var result = await mediator.Send(new GetPricingByIdQuery(id));
        
        return result.IsSuccess 
            ? Results.Ok(result.Value) 
            : Results.NotFound();
    }

    private static async Task<IResult> CreatePricing(
        CreatePricingRequest request, 
        IMediator mediator)
    {
        var command = new CreatePricingCommand(
            request.PlanType, 
            request.Price, 
            request.Features
        );
        
        var result = await mediator.Send(command);
        
        return result.IsSuccess 
            ? Results.Created($"/api/pricing/{result.Value}", result.Value)
            : Results.BadRequest(result.ErrorMessage);
    }
}

public record CreatePricingRequest(string PlanType, decimal Price, List<string> Features);

2. Domain Layer
Location: Domain/
Contains: Entities, Value Objects, Repository Interfaces

namespace MudBlazorWeb.Features.Pricing.Domain;

public class Pricing
{
    public Guid Id { get; private set; }
    public string PlanType { get; private set; }
    public decimal Price { get; private set; }
    public List<string> Features { get; private set; }

    private Pricing() { }

    public static Pricing Create(string planType, decimal price, List<string> features)
    {
        if (string.IsNullOrWhiteSpace(planType))
            throw new ArgumentException("Plan type cannot be empty");

        if (price < 0)
            throw new ArgumentException("Price cannot be negative");

        return new Pricing
        {
            Id = Guid.NewGuid(),
            PlanType = planType,
            Price = price,
            Features = features ?? new List<string>()
        };
    }
}

namespace MudBlazorWeb.Features.Pricing.Domain;

public interface IPricingRepository
{
    Task<List<Pricing>> GetAllAsync(CancellationToken cancellationToken = default);
    Task<Pricing?> GetByIdAsync(Guid id, CancellationToken cancellationToken = default);
    Task AddAsync(Pricing pricing, CancellationToken cancellationToken = default);
}

3. Infrastructure Layer
Location: Infrastructure/
Contains: Repository implementations, External services

using MudBlazorWeb.Features.Pricing.Domain;

namespace MudBlazorWeb.Features.Pricing.Infrastructure;

public class PricingRepository : IPricingRepository
{
    private readonly HttpClient _httpClient;

    public PricingRepository(HttpClient httpClient)
    {
        _httpClient = httpClient;
    }

    public async Task<List<Pricing>> GetAllAsync(CancellationToken cancellationToken = default)
    {
        // Example: Call API
        var response = await _httpClient.GetFromJsonAsync<List<PricingDto>>(
            "api/pricing", 
            cancellationToken
        );

        return response?.Select(dto => 
            Pricing.Create(dto.PlanType, dto.Price, dto.Features)
        ).ToList() ?? new List<Pricing>();
    }

    public async Task<Pricing?> GetByIdAsync(Guid id, CancellationToken cancellationToken = default)
    {
        var response = await _httpClient.GetFromJsonAsync<PricingDto>(
            $"api/pricing/{id}", 
            cancellationToken
        );

        return response != null 
            ? Pricing.Create(response.PlanType, response.Price, response.Features)
            : null;
    }

    public async Task AddAsync(Pricing pricing, CancellationToken cancellationToken = default)
    {
        await _httpClient.PostAsJsonAsync("api/pricing", pricing, cancellationToken);
    }
}

namespace MudBlazorWeb.Features.Pricing.UI;

public record PricingState
{
    public bool IsLoading { get; init; }
    public List<PricingDto> Pricings { get; init; } = new();
    public string? ErrorMessage { get; init; }
}

namespace MudBlazorWeb.Features.Pricing.UI;

public record LoadPricingsAction;
public record LoadPricingsSuccessAction(List<PricingDto> Pricings);
public record LoadPricingsFailureAction(string ErrorMessage);

using Fluxor;

namespace MudBlazorWeb.Features.Pricing.UI;

public static class PricingReducers
{
    [ReducerMethod]
    public static PricingState ReduceLoadPricingsAction(PricingState state, LoadPricingsAction action) =>
        state with { IsLoading = true, ErrorMessage = null };

    [ReducerMethod]
    public static PricingState ReduceLoadPricingsSuccessAction(
        PricingState state, 
        LoadPricingsSuccessAction action) =>
        state with { IsLoading = false, Pricings = action.Pricings };

    [ReducerMethod]
    public static PricingState ReduceLoadPricingsFailureAction(
        PricingState state, 
        LoadPricingsFailureAction action) =>
        state with { IsLoading = false, ErrorMessage = action.ErrorMessage };
}

using Fluxor;
using MediatR;
using MudBlazorWeb.Features.Pricing.Application;

namespace MudBlazorWeb.Features.Pricing.UI;

public class PricingEffects
{
    private readonly IMediator _mediator;

    public PricingEffects(IMediator mediator)
    {
        _mediator = mediator;
    }

    [EffectMethod]
    public async Task HandleLoadPricingsAction(LoadPricingsAction action, IDispatcher dispatcher)
    {
        var result = await _mediator.Send(new GetPricingsQuery());

        if (result.IsSuccess)
            dispatcher.Dispatch(new LoadPricingsSuccessAction(result.Value));
        else
            dispatcher.Dispatch(new LoadPricingsFailureAction(result.ErrorMessage));
    }
}

Single DependencyInjection.cs
One file per feature at feature root:

using Microsoft.Extensions.DependencyInjection;
using MudBlazorWeb.Features.Pricing.Domain;
using MudBlazorWeb.Features.Pricing.Infrastructure;

namespace MudBlazorWeb.Features.Pricing;

public static class DependencyInjection
{
    public static IServiceCollection AddPricingFeature(this IServiceCollection services)
    {
        // Application (MediatR auto-scans handlers)
        services.AddMediatR(cfg => 
            cfg.RegisterServicesFromAssembly(typeof(DependencyInjection).Assembly));
        
        // Infrastructure
        services.AddScoped<IPricingRepository, PricingRepository>();
        
        return services;
    }

    // Optional: Extension for endpoint mapping
    public static IEndpointRouteBuilder MapPricingFeature(this IEndpointRouteBuilder endpoints)
    {
        endpoints.MapPricingEndpoints();
        return endpoints;
    }
}

Implementation Details
1. Shared Infrastructure Layer
using Microsoft.EntityFrameworkCore;
using MudBlazorWeb.Features.Pricing.Domain;
using MudBlazorWeb.Features.Authentication.Domain;

namespace MudBlazorWeb.Infrastructure.Database;

public class AppDbContext : DbContext
{
    public AppDbContext(DbContextOptions<AppDbContext> options) : base(options) { }

    // DbSets for each feature
    public DbSet<Pricing> Pricings => Set<Pricing>();
    public DbSet<User> Users => Set<User>();

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        base.OnModelCreating(modelBuilder);

        // Apply all configurations from assemblies
        // This automatically finds all IEntityTypeConfiguration<T> implementations
        modelBuilder.ApplyConfigurationsFromAssembly(typeof(AppDbContext).Assembly);

        // Or apply specific feature configurations manually
        // modelBuilder.ApplyConfiguration(new PricingEntityConfiguration());
        // modelBuilder.ApplyConfiguration(new UserEntityConfiguration());
    }

    public override Task<int> SaveChangesAsync(CancellationToken cancellationToken = default)
    {
        // Handle audit fields
        var entries = ChangeTracker.Entries<BaseAuditableEntity>();

        foreach (var entry in entries)
        {
            switch (entry.State)
            {
                case EntityState.Added:
                    entry.Entity.CreatedOn = DateTime.UtcNow;
                    entry.Entity.CreatedBy = "System"; // Get from auth context
                    break;

                case EntityState.Modified:
                    entry.Entity.ModifiedOn = DateTime.UtcNow;
                    entry.Entity.ModifiedBy = "System"; // Get from auth context
                    break;

                case EntityState.Deleted:
                    entry.State = EntityState.Modified;
                    entry.Entity.DeletedOn = DateTime.UtcNow;
                    entry.Entity.DeletedBy = "System"; // Get from auth context
                    entry.Entity.IsDeleted = true;
                    break;
            }
        }

        return base.SaveChangesAsync(cancellationToken);
    }
}

using Microsoft.EntityFrameworkCore;

using Microsoft.Extensions.DependencyInjection;

namespace MudBlazorWeb.Infrastructure.Database;

public static class DependencyInjection
{
    public static IServiceCollection AddDatabase(
        this IServiceCollection services,
        IConfiguration configuration)
    {
        var connectionString = configuration.GetConnectionString("DefaultConnection")
            ?? throw new InvalidOperationException("Connection string 'DefaultConnection' not found.");

        services.AddDbContext<AppDbContext>(options =>
        {
            options.UseNpgsql(connectionString, npgsqlOptions =>
            {
                npgsqlOptions.MigrationsAssembly(typeof(AppDbContext).Assembly.FullName);
                npgsqlOptions.EnableRetryOnFailure(maxRetryCount: 3);
            });

#if DEBUG
            options.EnableSensitiveDataLogging();
            options.EnableDetailedErrors();
#endif
        });

        return services;
    }
}

using MudBlazorWeb.Infrastructure.Database;

namespace MudBlazorWeb.Features.Pricing.Domain;

public class Pricing : BaseAuditableEntity
{

    public decimal Price { get; private set; }

    public static Pricing Create(string planType, decimal price, List<string> features)
    {
        if (string.IsNullOrWhiteSpace(planType))
            throw new ArgumentException("Plan type cannot be empty");

        if (price < 0)
            throw new ArgumentException("Price cannot be negative");

        return new Pricing
        {
            Id = Guid.NewGuid(),
            PlanType = planType,
            Price = price,
            Features = features ?? new List<string>()
        };
    }

    public void UpdatePrice(decimal newPrice)
    {
        if (newPrice < 0)
            throw new ArgumentException("Price cannot be negative");

        Price = newPrice;
    }

using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Metadata.Builders;
using MudBlazorWeb.Features.Pricing.Domain;

namespace MudBlazorWeb.Features.Pricing.Infrastructure;

public class PricingEntityConfiguration : IEntityTypeConfiguration<Pricing>
{
    public void Configure(EntityTypeBuilder<Pricing> builder)
    {
        builder.ToTable("Pricings", schema: "pricing");

        builder.HasKey(p => p.Id);

        builder.Property(p => p.PlanType)
            .IsRequired()
            .HasMaxLength(100);

        builder.Property(p => p.Price)
            .IsRequired()
            .HasPrecision(18, 2);

        builder.Property(p => p.Features)
            .HasConversion(
                v => string.Join(',', v),
                v => v.Split(',', StringSplitOptions.RemoveEmptyEntries).ToList()
            )
            .HasMaxLength(1000);

        // Index
        builder.HasIndex(p => p.PlanType);

        // Query filter for soft delete
        builder.HasQueryFilter(p => !p.IsDeleted);

        // Seed data
        builder.HasData(
            Pricing.Create("Basic", 9.99m, new List<string> { "Feature 1", "Feature 2" }),
            Pricing.Create("Pro", 19.99m, new List<string> { "Feature 1", "Feature 2", "Feature 3" }),
            Pricing.Create("Enterprise", 49.99m, new List<string> { "All Features" })
        );
    }
}

namespace MudBlazorWeb.Features.Pricing.Infrastructure;

public class PricingRepository : IPricingRepository
{
    private readonly AppDbContext _context;

    public PricingRepository(AppDbContext context)
    {
        _context = context;
    }

    public async Task<List<Pricing>> GetAllAsync(CancellationToken cancellationToken = default)
    {
        return await _context.Pricings
            .OrderBy(p => p.Price)
            .ToListAsync(cancellationToken);
    }

    public async Task<Pricing?> GetByIdAsync(Guid id, CancellationToken cancellationToken = default)
    {
        return await _context.Pricings
            .FirstOrDefaultAsync(p => p.Id == id, cancellationToken);
    }

    public async Task AddAsync(Pricing pricing, CancellationToken cancellationToken = default)
    {
        await _context.Pricings.AddAsync(pricing, cancellationToken);
        await _context.SaveChangesAsync(cancellationToken);
    }

    public async Task UpdateAsync(Pricing pricing, CancellationToken cancellationToken = default)
    {
        _context.Pricings.Update(pricing);
        await _context.SaveChangesAsync(cancellationToken);
    }

    public async Task DeleteAsync(Pricing pricing, CancellationToken cancellationToken = default)
    {
        _context.Pricings.Remove(pricing);
        await _context.SaveChangesAsync(cancellationToken);
    }
}

Summary
✅ Recommended Structure (Shared DbContext)

Infrastructure/
├── Database/
│   ├── AppDbContext.cs              # ⭐ Shared DbContext
│   ├── BaseAuditableEntity.cs       # Shared base entity
│   ├── DependencyInjection.cs       # DbContext registration
│   └── Migrations/                  # All migrations

Features/Pricing/
├── Domain/
│   └── Pricing.cs                   # Entity (inherits BaseAuditableEntity)
├── Infrastructure/
│   ├── PricingEntityConfiguration.cs # ⭐ EF Configuration
│   └── PricingRepository.cs         # Uses AppDbContext
└── DependencyInjection.cs           # Registers repository only

Key Points
1.	Shared DbContext in Infrastructure/Database/
2.	Entity Configurations in each feature's Infrastructure/
3.	Migrations in shared Infrastructure/Database/Migrations/
4.	Repositories inject AppDbContext
5.	Register Database FIRST in Program.cs before features
This approach maintains Clean Architecture boundaries while keeping EF Core migrations manageable! 🎯
